Problem 1:
Given a array of positive and negative numbers. Write a program to rearrange the array such that the positive and negative numbers comes in alternate fashion.
Solution Code:
#include <stdio.h>
int main(){
    int n;
    printf("Enter the size of the array: ");
    scanf("%d", &n);
    int arr[n];
    for(int i = 0; i < n; i++){
        scanf("%d", &arr[i]);}
    int arr1[n] , arr2[n];
    int i = 0, j = 0, k = 0;
    while(i<n){
        if(arr[i] > 0){ arr1[j] = arr[i]; j++;}
        else{ arr2[k] = arr[i];  k++; }
        i++;}
    i = 0, j = 0, k = 0;
    while(k<n){ arr[k] = arr1[i] ; k++;i++;
                arr[k] = arr2[j] ; k++; j++;}
    printf("The new array is: \n");
    for (int i = 0; i < n; i++) {
        printf("%d  " , arr[i]);}
return 0;
}

Problem 2:
Write a program to print all subsets of array (which contains two elements) whose sum is equivalent to given number.
Solution Code:
#include <stdio.h>
int main(){
    int n;
    printf("Enter the size of the array: ");
    scanf("%d", &n);
    int arr[n];
    printf("Enter elements of the array: ");
    for(int i = 0; i < n; i++){
        scanf("%d", &arr[i]);}
    int target;
    printf("Enter the target sum: ");
    scanf("%d", &target);
    for(int i = 0; i < n; i++){
      for(int j = i + 1; j < n; j++){
        if(arr[i]+arr[j]==target)  printf("Pair: %d, %d  ", arr[i], arr[j]);
      }
    }
return 0;
}

Problem 3:
Write a program to input an array of structures for employees and store it in a file one structure at a time  , also read the salary input of file of every employ and update in the file using rules:
1)	Salary > 50000 – increment of 5% in salary
2)	Salary > 20000 – increment of 10% in salary
3)	Otherwise – increment of 15% in salary
Solution Code:
#include <stdio.h>
struct emp{
    int empid;
    char name[20];
    float salary;
};
int main(){
    struct emp emp[5];
    for(int i=0;i<5;i++){
        printf("Enter employee ID of emp%d: ",i+1);
        scanf("%d",&emp[i].empid);
        printf("Enter name of emp%d: ",i+1);
        scanf("%s",&emp[i].name);
        printf("Enter salary of emp%d: ",i+1);
        scanf("%f",&emp[i].salary);
    }
    FILE *fp;
    fp=fopen("lab1.dat","wb");
    for(int j=0;j<5;j++){
        fwrite(&emp[j],sizeof(struct emp),1,fp);
    }
    fclose(fp);
    fp=fopen("lab1.dat","rb");
    for(int i=0;i<5;i++){
        fread(&emp[i], sizeof(struct emp), 1, fp);
        if(emp[i].salary>=50000) emp[i].salary+=(emp[i].salary*0.05);
        else if(emp[i].salary>=20000 && emp[i].salary<50000) emp[i].salary += (emp[i].salary*0.1); 
        else emp[i].salary+=(emp[i].salary*0.15);
    }
    fclose(fp);
    fp=fopen("lab1.dat","wb");
    for (int i=0;i<5;i++){
        fwrite(&emp[i],sizeof(struct emp),1,fp);
    }
    fclose(fp);
    fp=fopen("lab1.dat","rb");
    for(int i=0;i<5;i++){
        fread(&emp[i], sizeof(struct emp), 1, fp);
        printf("Data of emp%d: %d %s %.2f\n",i+1,emp[i].empid,emp[i].name,emp[i].salary);
    }
    fclose(fp);
    return 0;
}
